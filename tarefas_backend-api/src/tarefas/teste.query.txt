import { Controller, Get, Query } from '@nestjs/common';

@Controller('produtos')
export class ProdutosController {

  // A rota base é apenas GET /produtos
  @Get()
  listarProdutos(
    @Query('categoria') categoria?: string, // Opcional
    @Query('limite') limite: number = 10,   // Opcional, com valor padrão 10
  ) {
    // Se a categoria foi informada na URL, aplicamos o filtro
    if (categoria) {
      return {
        mensagem: `Listando até ${limite} produtos da categoria "${categoria}"`,
        filtrosAplicados: { categoria, limite },
      };
    }

    // Se nenhuma categoria foi passada, lista os produtos no geral
    return {
      mensagem: `Listando os primeiros ${limite} produtos cadastrados`,
      filtrosAplicados: { categoria: 'todas', limite },
    };
  }
}


// Como o cliente pode chamar:

// Sem filtros: GET http://localhost:3000/produtos

// Resultado: Retorna produtos no geral (limite padrão 10).

// Com 1 filtro: GET http://localhost:3000/produtos?categoria=eletronicos

// Resultado: Retorna apenas eletrônicos.

// Com múltiplos filtros: GET http://localhost:3000/produtos?categoria=eletronicos&limite=5

// Resultado: Retorna 5 eletrônicos.
